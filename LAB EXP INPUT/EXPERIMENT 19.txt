import pandas as pd
from sklearn.naive_bayes import GaussianNB

data = {
    'Income': [20000, 30000, 50000, 60000, 80000, 90000],
    'Credit': [500, 550, 650, 700, 750, 800],
    'Loan': [15000, 20000, 25000, 30000, 35000, 40000],
    'Approved': [0, 0, 1, 1, 1, 1]
}

df = pd.DataFrame(data)

X = df[['Income', 'Credit', 'Loan']]
y = df['Approved']

model = GaussianNB()
model.fit(X, y)

prediction = model.predict([[55000, 680, 25000]])

if prediction[0] == 1:
    print("Loan Status: Approved")
else:
    print("Loan Status: Not Approved")
