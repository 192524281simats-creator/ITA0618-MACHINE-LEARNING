import numpy as np
from sklearn.mixture import GaussianMixture

X = np.array([
    [1, 2], [1, 3], [2, 2],
    [8, 8], [9, 8], [8, 9]
])

model = GaussianMixture(
    n_components=2,
    random_state=42
)

model.fit(X)

labels = model.predict(X)

print("Cluster Labels:", labels)
print("Cluster Centers:")
print(model.means_)
