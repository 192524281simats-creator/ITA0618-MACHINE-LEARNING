import pandas as pd
from sklearn.tree import DecisionTreeClassifier

data = {
    'Income': [20000, 50000, 80000, 30000, 90000, 40000],
    'Loan': [10000, 20000, 30000, 15000, 25000, 18000],
    'Payment': [1, 2, 3, 1, 3, 2],
    'Score': ['Poor', 'Good', 'Excellent',
              'Poor', 'Excellent', 'Good']
}
df = pd.DataFrame(data)

X = df[['Income', 'Loan', 'Payment']]
y = df['Score']

model = DecisionTreeClassifier()
model.fit(X, y)

prediction = model.predict([[60000, 20000, 2]])

print("Credit Score:", prediction[0])
