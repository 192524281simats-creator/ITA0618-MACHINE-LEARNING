import pandas as pd
from sklearn.linear_model import LinearRegression

data = {
    'Year': [2015, 2016, 2017, 2018, 2019, 2020],
    'Mileage': [50000, 45000, 40000, 35000, 30000, 25000],
    'Engine': [1000, 1200, 1200, 1500, 1500, 1800],
    'Price': [300000, 350000, 400000, 450000, 500000, 600000]
}

df = pd.DataFrame(data)

X = df[['Year', 'Mileage', 'Engine']]
y = df['Price']

model = LinearRegression()
model.fit(X, y)

prediction = model.predict([[2021, 20000, 1800]])

print("Predicted Car Price: Rs.", round(prediction[0], 2))
